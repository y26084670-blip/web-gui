********** уравнение многофазной эл цепи ***********
3	число ветвей
1 	1	0.0	0.0	1.txt
2 	2	0.0	0.0	2.txt
3 	3	0.0	0.0	3.txt
1	тип соединения: 0-с нулевым проводом, 1-без нулевого провода
220.0	номинальная амплитуда ЭДС (Вольт)
1	тип материала витков (0-материал без потерь, 1-ВТСП)
5.0	абс погрешность норvы невязки (Вольт)
100.0 	шаг по току при вычислении производных (Ампер)
7	условный номер метода фортран-библиотеки оптимизатора (1..11)

*** пояснение к эл цепи ***
для каждой ветви задается строка значений через пробел и/или табуляцию следующих элементов:
-номер искомого импульса, описывающего ток в цепи (изначально нули для каждого момента времени)
-номер измерительной катушки в эл цепи (или 0, если нет катушки)
-значение постоянной индуктивности, Гн
-значение постоянного активного сопротивления, Ом
-имя файла в каталоге задачи с временными узлами аппроксимации нормированного напряжения как функции времени (если файл не найден, то - нулевая ЭДС в данной фазе)

примечание1: для создания файлов для ЭДС в ветвях трехфазной цепи можно использовать
\CLARK.MPI.START.204\addons\Circuit\circuit_utils.exe
В диалоге записывает в отдельный файл для каждой из 3-х фаз таблицу НОРМИРОВАННЫХ ЭДС для заданных:
шаг по времени (сек)
число шагов
частота (Гц)
интервал времени (сек) exp-вхождения в номинальный 3-х фазный режим от 0. Если 0.0, то без плавного вхождения
Каждая последующая  фаза отстает на 120 град.

примечание2: режимы фортран-оптимизатора
n_intervals = 1
bounds_mode = NLESOLVER_IGNORE_BOUNDS
norm_mode = NLESOLVER_2_NORM

    select case (CIRC.ivar)
    case( 1)  ! 'Constant alpha'
        step_mode = 1
        use_broyden = .false.
    case( 2)  ! 'Constant alpha + broyden'
        step_mode = 1
        use_broyden = .true.
    case( 3)  ! 'Backtracking line search'
        step_mode = 2
        use_broyden = .false.
    case( 4)  ! 'Backtracking line search + broyden'
        step_mode = 2
        use_broyden = .true.
    case( 5)  ! 'Exact line search'
        step_mode = 3
        use_broyden = .false.
    case( 6)  ! 'Exact line search + broyden'
        step_mode = 3
        use_broyden = .true.
    case( 7)  ! 'Fixed point search'
        step_mode = 4
        use_broyden = .false.
    case( 8)  ! 'Fixed point search + broyden'
        step_mode = 4
        use_broyden = .true.
    case( 9)  ! 'Fixed point search + broyden + scalar bounds'
        step_mode = 4
        use_broyden = .true.
        bounds_mode = NLESOLVER_SCALAR_BOUNDS
    case(10)  ! 'Fixed point search + broyden + vector bounds'
        step_mode = 4
        use_broyden = .true.
        bounds_mode = NLESOLVER_VECTOR_BOUNDS
    case(11)  ! 'Fixed point search + broyden + wall bounds'
        step_mode = 4
        use_broyden = .true.
        bounds_mode = NLESOLVER_WALL_BOUNDS
    end select
